cmake_minimum_required(VERSION 3.10.0)
set(ProjectName "SFML_TEMPLATE")
project(ProjectName)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SFML_STATIC_LIBRARIES TRUE)
set(SFML_DIR "$ENV{HOME}/lib/lib/cmake/SFML")
find_package(SFML 3.0.0 COMPONENTS System Window Graphics REQUIRED)
add_executable(${ProjectName} main.cpp)
target_include_directories(${ProjectName} PRIVATE $ENV{HOME}/lib/include)
target_compile_definitions(${ProjectName} PRIVATE SFML_STATIC)
target_link_libraries(${ProjectName} SFML::Graphics SFML::Window SFML::System)
